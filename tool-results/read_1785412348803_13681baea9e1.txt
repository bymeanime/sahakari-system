   103→   103→function BSDatePicker({ value, onChange, label }: { value: string; onChange: (v: string) => void; label?: string }) {
   104→   104→  const [open, setOpen] = useState(false)
   105→   105→  const parts = value ? value.split('-').map(Number) : [2082, 1, 1]
   106→   106→  const [year, setYear] = useState(parts[0])
   107→   107→  const [month, setMonth] = useState(parts[1])
   108→   108→  const [day, setDay] = useState(parts[2])
   109→   109→  const years = getBSYearRange()
   110→   110→
   111→   111→  const formatted = value ? formatBSDate(value) : { en: 'Select Date', nep: 'मिति छान्नुहोस्' }
   112→   112→  const grid = getBSMonthGrid(year, month)
   113→   113→  const maxDays = getBSMonthDays(year, month)
   114→   114→
   115→   115→  const selectDay = (d: number) => {
   116→   116→    const dateStr = `${year}-${String(month).padStart(2, '0')}-${String(d).padStart(2, '0')}`
   117→   117→    onChange(dateStr)
   118→   118→    setOpen(false)
   119→   119→  }
   120→   120→
   121→   121→  return (
   122→   122→    <div className="space-y-1">
   123→   123→      {label && <Label className="text-xs">{label}</Label>}
   124→   124→      <div className="relative">
   125→   125→        <button
   126→   126→          onClick={() => setOpen(!open)}
   127→   127→          className="w-full flex items-center justify-between px-3 py-2 border rounded-lg bg-white text-sm text-left hover:border-emerald-400 transition-colors"
   128→   128→        >
   129→   129→          <div>
   130→   130→            <span className="text-gray-900">{formatted.en}</span>
   131→   131→            <span className="text-gray-400 ml-2 text-xs">({formatted.nep})</span>
   132→   132→          </div>
   133→   133→          <CalendarDays className="w-4 h-4 text-gray-400" />
   134→   134→        </button>
   135→   135→        {open && (
   136→   136→          <div className="absolute top-full mt-1 left-0 z-50 bg-white border rounded-xl shadow-xl p-4 w-80">
   137→   137→            <div className="flex items-center gap-2 mb-3">
   138→   138→              <Select value={String(year)} onValueChange={v => { setYear(Number(v)); setDay(1) }}>
   139→   139→                <SelectTrigger className="w-24 h-8 text-xs"><SelectValue /></SelectTrigger>
   140→   140→                <SelectContent>
   141→   141→                  {years.map(y => <SelectItem key={y} value={String(y)}>{toNepaliDigits(y)}</SelectItem>)}
   142→   142→                </SelectContent>
   143→   143→              </Select>
   144→   144→              <Select value={String(month)} onValueChange={v => { setMonth(Number(v)); setDay(1) }}>
   145→   145→                <SelectTrigger className="w-28 h-8 text-xs"><SelectValue /></SelectTrigger>
   146→   146→                <SelectContent>
   147→   147→                  {bsMonths.map((m, i) => <SelectItem key={i} value={String(i + 1)}>{m} ({bsMonthsNep[i]})</SelectItem>)}
   148→   148→                </SelectContent>
   149→   149→              </Select>
   150→   150→            </div>
   151→   151→            <div className="grid grid-cols-7 gap-0.5 text-center mb-1">
   152→   152→              {['आ', 'सो', 'मं', 'बु', 'बि', 'शु', 'श'].map((d, i) => (
   153→   153→                <div key={i} className="text-xs text-gray-400 py-1 font-medium">{d}</div>
   154→   154→              ))}
   155→   155→            </div>
   156→   156→            <div className="grid grid-cols-7 gap-0.5">
   157→   157→              {grid.flat().map((d, i) => (
   158→   158→                <button
   159→   159→                  key={i}
   160→   160→                  disabled={!d}
   161→   161→                  onClick={() => d && selectDay(d)}
   162→   162→                  className={`h-8 w-8 rounded-lg text-xs flex items-center justify-center transition-colors ${
   163→   163→                    d === day && month === parts[1] && year === parts[0]
   164→   164→                      ? 'bg-emerald-600 text-white font-bold'
   165→   165→                      : d ? 'hover:bg-emerald-50 text-gray-700' : 'text-transparent'
   166→   166→                  }`}
   167→   167→                >
   168→   168→                  {d ? toNepaliDigits(d) : ''}
   169→   169→                </button>
   170→   170→              ))}
   171→   171→            </div>
   172→   172→            <div className="mt-3 pt-2 border-t flex justify-between items-center">
   173→   173→              <span className="text-xs text-gray-400">{bsMonths[month - 1]} {year} - {toNepaliDigits(maxDays)} days</span>
   174→   174→              <Button size="sm" variant="ghost" className="text-xs h-7" onClick={() => { onChange(getTodayBS()); setOpen(false) }}>Today</Button>

   176→   176→          </div>
   177→   177→        )}
   178→   178→      </div>

   180→   180→  )

   193→   193→        .then(r => r.json())
   194→   194→        .then(d => { setNotifications(Array.isArray(d) ? d : []) })
