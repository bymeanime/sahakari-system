    32→  MIN_CAPITAL_ADEQUACY_RATIO: 8,

    44→    SUB_STANDARD: { label: 'Sub-standard', labelNep: 'उप-मानक', provision: 25, provisionNep: '२५%' },
    45→    DOUBTFUL: { label: 'Doubtful', labelNep: 'शंकास्पद', provision: 50, provisionNep: '५०%' },

    63→function formatNPR(amount: number): string {
    64→  return `Rs. ${amount.toLocaleString('en-NP', { minimumFractionDigits: 2, maximumFractionDigits: 2 })}`
    65→}
    66→
    67→function toNepaliDigits(num: number | string): string {
    68→  const nepaliDigits = ['०', '१', '२', '३', '४', '५', '६', '७', '८', '९']
    69→  return String(num).replace(/[0-9]/g, (d) => nepaliDigits[parseInt(d)])
    70→}
    71→
    72→function formatNPRNepali(amount: number): string {
    73→  const formatted = amount.toLocaleString('en-NP', { minimumFractionDigits: 2, maximumFractionDigits: 2 })
    74→  return `रु. ${toNepaliDigits(formatted)}`
    75→}

    77→/** Get current fiscal year in BS format */
    78→function getCurrentFiscalYear(): string {
    79→  const today = new Date()
    80→  const bsYear = today.getFullYear() + 57
    81→  // BS fiscal year: if current month is after Chaitra (mid-April), next FY
    82→  const fiscalStart = today.getMonth() + 1 > 3 ? bsYear : bsYear - 1
    83→  return `${fiscalStart}/${(fiscalStart + 1) % 100}`
    84→}
    85→
    86→/** Get previous fiscal year */
    87→function getPreviousFiscalYear(): string {
    88→  const current = getCurrentFiscalYear()
    89→  const [startYear] = current.split('/').map(Number)
    90→  return `${startYear - 1}/${startYear % 100}`
    91→}

    93→/** Classify NPL based on days past due */
    94→function classifyNPL(daysPastDue: number): keyof typeof NRB_COMPLIANCE.NPL_CLASSIFICATION {
    95→  if (daysPastDue <= NRB_COMPLIANCE.NPL_THRESHOLDS.PASS) return 'PASS'
    96→  if (daysPastDue <= NRB_COMPLIANCE.NPL_THRESHOLDS.WATCH_LIST) return 'WATCH_LIST'
    97→  if (daysPastDue <= NRB_COMPLIANCE.NPL_THRESHOLDS.SUB_STANDARD) return 'SUB_STANDARD'
    98→  if (daysPastDue <= NRB_COMPLIANCE.NPL_THRESHOLDS.DOUBTFUL) return 'DOUBTFUL'
    99→  return 'LOSS'
   100→}
   101→
   102→/** Calculate days past due from a due date string */
   103→function getDaysPastDue(dueDate: string): number {
   104→  const due = new Date(dueDate)
   105→  const now = new Date()
   106→  const diffMs = now.getTime() - due.getTime()
   107→  return Math.max(0, Math.floor(diffMs / (1000 * 60 * 60 * 24)))
   108→}

   224→// 1. Balance Sheet (निश्शेष जानकारी)
   225→// -----------------------------------------------------------
   226→
   227→function generateBalanceSheet(data: Awaited<ReturnType<typeof fetchReportData>>) {
   228→  const header = getInstitutionHeader(data.organization)
   229→
   230→  // Calculate totals
   231→  const totalSavingsDeposits = data.savingsAccounts.reduce((s, a) => s + a.balance, 0)
   232→  const totalShareCapital = data.shareHoldings.reduce((s, sh) => s + sh.shareValue, 0)
   233→  const totalLoanReceivable = data.loanApplications
   234→    .filter((l) => l.status === 'DISBURSED')
   235→    .reduce((s, l) => s + (l.outstandingAmount || 0), 0)
   236→  const totalFixedAssets = data.assets.reduce((s, a) => s + a.currentValue, 0)
   237→  const totalDepreciation = data.assets.reduce((s, a) => s + a.accumulatedDep, 0)
   238→  const netFixedAssets = totalFixedAssets - totalDepreciation
   239→
   240→  // Accounting totals from journal entries
   241→  const totalDebits = data.journalEntries.reduce((sum, je) => sum + je.items.reduce((s, i) => s + i.debit, 0), 0)
   242→  const totalCredits = data.journalEntries.reduce((sum, je) => sum + je.items.reduce((s, i) => s + i.credit, 0), 0)
   243→
   244→  // Income & Expense for retained earnings
   245→  const interestIncome = data.loanApplications
   246→    .filter((l) => l.status === 'DISBURSED')
   247→    .reduce((s, l) => s + ((l.disbursedAmount || 0) * (l.interestRate || 0)) / 100 / 12, 0)
   248→  const interestExpense = data.savingsAccounts.reduce((s, a) => s + (a.balance * (a.product?.interestRate || 0)) / 100 / 12, 0)
   249→  const totalSalaryExpense = data.employees.reduce((s, e) => s + e.salary, 0)
   250→  const netIncome = interestIncome - interestExpense - totalSalaryExpense
   251→
   252→  // Cash & bank balance from accounts
   253→  const cashAccounts = data.accounts.filter((a) => a.type === 'ASSET' && (a.subType === 'CASH' || a.subType === 'BANK'))
   254→  const cashInHand = cashAccounts.filter((a) => a.subType === 'CASH').reduce((s, a) => {
   255→    const debitTotal = data.journalEntryItems.filter((i) => i.accountId === a.id).reduce((s, i) => s + i.debit - i.credit, 0)
   256→    return s + Math.max(0, debitTotal)
   257→  }, 0)
   258→  const bankBalance = cashAccounts.filter((a) => a.subType === 'BANK').reduce((s, a) => {
   259→    const debitTotal = data.journalEntryItems.filter((i) => i.accountId === a.id).reduce((s, i) => s + i.debit - i.credit, 0)
   260→    return s + Math.max(0, debitTotal)
   261→  }, 0)
   262→
   263→  // Reserve fund (from equity accounts)
   264→  const reserveFund = data.accounts
   265→    .filter((a) => a.type === 'EQUITY' && a.name.toLowerCase().includes('reserve'))
   266→    .reduce((s, a) => {
   267→      const creditTotal = data.journalEntryItems.filter((i) => i.accountId === a.id).reduce((s, i) => s + i.credit - i.debit, 0)
   268→      return s + Math.max(0, creditTotal)
   269→    }, 0)
   270→
   271→  const totalAssets = cashInHand + bankBalance + totalLoanReceivable + netFixedAssets
   272→  const totalLiabilities = totalSavingsDeposits + totalShareCapital + reserveFund + (netIncome > 0 ? netIncome : 0)
   273→
   274→  return {
   275→    reportType: 'balance-sheet',
   276→    reportTitle: 'Balance Sheet',
   277→    reportTitleNep: 'निश्शेष जानकारी',
   278→    header,
   279→    nrbCompliance: {
   280→      capitalAdequacyRatio: totalLoanReceivable > 0 ? ((totalShareCapital + reserveFund + (netIncome > 0 ? netIncome : 0)) / totalLoanReceivable) * 100 : 0,
   281→      minRequiredRatio: NRB_COMPLIANCE.MIN_CAPITAL_ADEQUACY_RATIO,
   282→      isCompliant: totalLoanReceivable > 0 ? ((totalShareCapital + reserveFund + (netIncome > 0 ? netIncome : 0)) / totalLoanReceivable) * 100 >= NRB_COMPLIANCE.MIN_CAPITAL_ADEQUACY_RATIO : true,
   283→    },
   284→    currentPeriod: {
   285→      assets: {
   286→        cashInHand: { amount: cashInHand, label: 'Cash in Hand', labelNep: 'हातमा नगद' },
   287→        bankBalance: { amount: bankBalance, label: 'Bank Balance', labelNep: 'बैंक शेष' },
   288→        loanReceivable: { amount: totalLoanReceivable, label: 'Loan Receivable', labelNep: 'ऋण उपलब्ध' },
   289→        fixedAssets: { amount: netFixedAssets, label: 'Fixed Assets (Net)', labelNep: 'स्थायी सम्पत्ति (शुद्ध)' },
   290→        totalAssets: { amount: totalAssets, label: 'Total Assets', labelNep: 'कुल सम्पत्ति' },
   291→      },
   292→      liabilities: {
   293→        memberDeposits: { amount: totalSavingsDeposits, label: 'Member Deposits', labelNep: 'सदस्य निक्षेप' },
   294→        shareCapital: { amount: totalShareCapital, label: 'Share Capital', labelNep: 'शेयर पूँजी' },
   295→        reserveFund: { amount: reserveFund, label: 'Reserve Fund', labelNep: 'आरक्ष कोष' },
   296→        retainedEarnings: { amount: netIncome > 0 ? netIncome : 0, label: 'Retained Earnings', labelNep: 'सञ्चित आय' },
   297→        totalLiabilities: { amount: totalLiabilities, label: 'Total Liabilities & Equity', labelNep: 'कुल दायित्व र पूँजी' },
   298→      },
   299→    },
   300→    previousPeriod: {
   301→      // Previous year comparison (placeholder - would need historical data)
   302→      note: 'Previous year figures require historical data from prior fiscal year records.',
   303→    },
   304→    summary: {
   305→      totalAssets,
   306→      totalLiabilities,
   307→      surplus: totalAssets - totalLiabilities,
   308→      isBalanced: Math.abs(totalAssets - totalLiabilities) < 1,
   309→    },
   310→    currency: 'NPR',
   311→    currencyNep: 'नेपाली रूपैयाँ',
   312→  }
   313→}

   316→// 2. Income Statement (आय-व्यय जानकारी)
   317→// -----------------------------------------------------------
   318→
   319→function generateIncomeStatement(data: Awaited<ReturnType<typeof fetchReportData>>) {
   320→  const header = getInstitutionHeader(data.organization)
   321→
   322→  // Income calculations
   323→  const interestIncome = data.loanApplications
   324→    .filter((l) => l.status === 'DISBURSED')
   325→    .reduce((s, l) => s + ((l.disbursedAmount || 0) * (l.interestRate || 0)) / 100 / 12, 0)
   326→  const feeIncome = data.loanApplications
   327→    .filter((l) => l.status !== 'PENDING')
   328→    .reduce((s, l) => s + (l.product?.processingFee || 0), 0)
   329→  const penaltyIncome = data.loanRepayments.reduce((s, r) => s + r.penaltyAmount, 0)
   330→
   331→  // Expense calculations
   332→  const interestExpense = data.savingsAccounts.reduce((s, a) => s + (a.balance * (a.product?.interestRate || 0)) / 100 / 12, 0)
   333→  const totalSalaryExpense = data.employees.reduce((s, e) => s + e.salary, 0)
   334→  const totalDepreciation = data.assets.reduce((s, a) => s + a.accumulatedDep, 0) / 5
   335→  const officeExpense = data.accounts
   336→    .filter((a) => a.type === 'EXPENSE' && a.name.toLowerCase().includes('office'))
   337→    .reduce((s, a) => {
   338→      const debitTotal = data.journalEntryItems.filter((i) => i.accountId === a.id).reduce((s, i) => s + i.debit, 0)
   339→      return s + debitTotal
   340→    }, 0)
   341→
   342→  const totalIncome = interestIncome + feeIncome + penaltyIncome
   343→  const totalExpenses = interestExpense + totalSalaryExpense + officeExpense + totalDepreciation
   344→  const netIncome = totalIncome - totalExpenses
   345→
   346→  return {
   347→    reportType: 'income-statement',
   348→    reportTitle: 'Income Statement',
   349→    reportTitleNep: 'आय-व्यय जानकारी',
   350→    header,
   351→    currentPeriod: {
   352→      income: {
   353→        interestIncome: { amount: Math.round(interestIncome), label: 'Interest Income', labelNep: 'ब्याज आय' },
   354→        feeIncome: { amount: Math.round(feeIncome), label: 'Fee & Commission Income', labelNep: 'शुल्क तथा कमिसन आय' },
   355→        penaltyIncome: { amount: Math.round(penaltyIncome), label: 'Penalty Income', labelNep: 'जरिवाना आय' },
   356→        otherIncome: { amount: 0, label: 'Other Income', labelNep: 'अन्य आय' },
   357→        totalIncome: { amount: Math.round(totalIncome), label: 'Total Income', labelNep: 'कुल आय' },
   358→      },
   359→      expenses: {
   360→        interestExpense: { amount: Math.round(interestExpense), label: 'Interest Expense', labelNep: 'ब्याज खर्च' },
   361→        salaryExpense: { amount: Math.round(totalSalaryExpense), label: 'Salary & Benefits', labelNep: 'तलब तथा भत्ता' },
   362→        officeExpense: { amount: Math.round(officeExpense), label: 'Office & Administrative', labelNep: 'कार्यालय तथा प्रशासनिक' },
   363→        depreciation: { amount: Math.round(totalDepreciation), label: 'Depreciation', labelNep: 'ह्रासकट्टी' },
   364→        otherExpenses: { amount: 0, label: 'Other Expenses', labelNep: 'अन्य खर्च' },
   365→        totalExpenses: { amount: Math.round(totalExpenses), label: 'Total Expenses', labelNep: 'कुल खर्च' },
   366→      },
   367→      netIncome: {
   368→        amount: Math.round(netIncome),
   369→        label: 'Net Income / (Loss)',
   370→        labelNep: 'खुद आय / (घाटा)',
   371→      },
   372→    },
   373→    previousPeriod: {
   374→      note: 'Previous year figures require historical data from prior fiscal year records.',
   375→    },
   376→    keyRatios: {
   377→      returnOnAssets: data.assets.reduce((s, a) => s + a.currentValue, 0) > 0
   378→        ? ((netIncome / data.assets.reduce((s, a) => s + a.currentValue, 0)) * 100).toFixed(2) + '%'
   379→        : 'N/A',
   380→      netInterestMargin: totalIncome > 0
   381→        ? (((interestIncome - interestExpense) / totalIncome) * 100).toFixed(2) + '%'
   382→        : 'N/A',
   383→      operatingRatio: totalIncome > 0
   384→        ? ((totalExpenses / totalIncome) * 100).toFixed(2) + '%'
   385→        : 'N/A',
   386→    },
   387→    currency: 'NPR',
   388→    currencyNep: 'नेपाली रूपैयाँ',
   389→  }
   390→}

   394→// -----------------------------------------------------------
   395→
   396→function generateNRBReturn(data: Awaited<ReturnType<typeof fetchReportData>>) {
   397→  const header = getInstitutionHeader(data.organization)
   398→
   399→  const totalSavingsDeposits = data.savingsAccounts.reduce((s, a) => s + a.balance, 0)
   400→  const totalShareCapital = data.shareHoldings.reduce((s, sh) => s + sh.shareValue, 0)
   401→  const totalLoansDisbursed = data.loanApplications
   402→    .filter((l) => l.status === 'DISBURSED')
   403→    .reduce((s, l) => s + (l.disbursedAmount || 0), 0)
   404→  const totalOutstanding = data.loanApplications
   405→    .filter((l) => l.status === 'DISBURSED')
   406→    .reduce((s, l) => s + (l.outstandingAmount || 0), 0)
   407→  const totalCollected = totalLoansDisbursed - totalOutstanding
   408→  const totalFixedAssets = data.assets.reduce((s, a) => s + a.currentValue, 0)
   409→  const totalSalaryExpense = data.employees.reduce((s, e) => s + e.salary, 0)
   410→  const interestIncome = data.loanApplications
   411→    .filter((l) => l.status === 'DISBURSED')
   412→    .reduce((s, l) => s + ((l.disbursedAmount || 0) * (l.interestRate || 0)) / 100 / 12, 0)
   413→  const interestExpense = data.savingsAccounts.reduce((s, a) => s + (a.balance * (a.product?.interestRate || 0)) / 100 / 12, 0)
   414→  const netIncome = interestIncome - interestExpense - totalSalaryExpense
   415→
   416→  // NPL classification
   417→  const overdueRepayments = data.loanRepayments.filter((r) => r.status === 'OVERDUE')
   418→  const nplByClassification: Record<string, { count: number; amount: number; provision: number }> = {
   419→    PASS: { count: 0, amount: 0, provision: 0 },
   420→    WATCH_LIST: { count: 0, amount: 0, provision: 0 },
   421→    SUB_STANDARD: { count: 0, amount: 0, provision: 0 },
   422→    DOUBTFUL: { count: 0, amount: 0, provision: 0 },
   423→    LOSS: { count: 0, amount: 0, provision: 0 },
   424→  }
   425→
   426→  // Classify overdue loans
   427→  const disbursedLoans = data.loanApplications.filter((l) => l.status === 'DISBURSED')
   428→  disbursedLoans.forEach((loan) => {

   440→    } else {

   442→      const outstanding = loan.outstandingAmount || 0

   444→      nplByClassification.PASS.amount += outstanding

   450→  const totalProvisions = Object.values(nplByClassification).reduce((s, c) => s + c.provision, 0)
   451→  const nplRatio = totalLoansDisbursed > 0 ? (totalNPL / totalLoansDisbursed) * 100 : 0
   452→
   453→  // Capital adequacy (includes mandatory reserve fund per Cooperative Act 2047 Section 66)
   454→  const reserveFund = netIncome > 0 ? Math.round(netIncome * 0.25) : 0
   455→  const capitalFund = totalShareCapital + reserveFund + (netIncome > 0 ? netIncome : 0)
   456→  const capitalAdequacyRatio = totalLoansDisbursed > 0 ? (capitalFund / totalLoansDisbursed) * 100 : 0
   457→
   458→  // Single borrower exposure
   459→  const memberLoanExposure = disbursedLoans.reduce((acc: Record<string, number>, loan) => {
   460→    const memberId = loan.memberId
   461→    acc[memberId] = (acc[memberId] || 0) + (loan.outstandingAmount || 0)
   462→    return acc
   463→  }, {})
   464→  const maxSingleExposure = Math.max(...Object.values(memberLoanExposure), 0)
   465→  const singleBorrowerRatio = capitalFund > 0 ? (maxSingleExposure / capitalFund) * 100 : 0
   466→
   467→  // Liquidity
   468→  const liquidAssets = totalSavingsDeposits * 0.15 // simplified
   469→  const liquidityRatio = totalSavingsDeposits > 0 ? (liquidAssets / totalSavingsDeposits) * 100 : 0
   470→
   471→  return {
   472→    reportType: 'nrb-return',
   473→    reportTitle: 'NRB Quarterly Return',

   475→    header,

   517→    liquidity: {

   543→      singleBorrowerExposureCompliant: singleBorrowerRatio <= NRB_COMPLIANCE.MAX_SINGLE_BORROWER_EXPOSURE,

   546→        && singleBorrowerRatio <= NRB_COMPLIANCE.MAX_SINGLE_BORROWER_EXPOSURE
   547→        && liquidityRatio >= NRB_COMPLIANCE.MIN_STATUTORY_LIQUIDITY_RATIO,

   557→
   558→function generateCapitalAdequacy(data: Awaited<ReturnType<typeof fetchReportData>>) {
   559→  const header = getInstitutionHeader(data.organization)
   560→
   561→  const totalShareCapital = data.shareHoldings.reduce((s, sh) => s + sh.shareValue, 0)
   562→  const totalLoansDisbursed = data.loanApplications
   563→    .filter((l) => l.status === 'DISBURSED')
   564→    .reduce((s, l) => s + (l.disbursedAmount || 0), 0)
   565→  const totalOutstanding = data.loanApplications
   566→    .filter((l) => l.status === 'DISBURSED')
   567→    .reduce((s, l) => s + (l.outstandingAmount || 0), 0)
   568→
   569→  const interestIncome = data.loanApplications
   570→    .filter((l) => l.status === 'DISBURSED')
   571→    .reduce((s, l) => s + ((l.disbursedAmount || 0) * (l.interestRate || 0)) / 100 / 12, 0)
   572→  const interestExpense = data.savingsAccounts.reduce((s, a) => s + (a.balance * (a.product?.interestRate || 0)) / 100 / 12, 0)
   573→  const totalSalaryExpense = data.employees.reduce((s, e) => s + e.salary, 0)
   574→  const netIncome = interestIncome - interestExpense - totalSalaryExpense
   575→
   576→  // Capital components (includes mandatory reserve fund per Cooperative Act 2047 Section 66)
   577→  const reserveFund = netIncome > 0 ? Math.round(netIncome * 0.25) : 0
   578→  const tier1Capital = totalShareCapital + reserveFund + (netIncome > 0 ? netIncome : 0)
   579→  const tier2Capital = 0 // Subordinated debt, etc. - not applicable for most cooperatives
   580→  const totalCapital = tier1Capital + tier2Capital
   581→
   582→  // Risk-weighted assets calculation
   583→  const riskWeights = {
   584→    cashAndBank: { weight: 0, label: 'Cash & Bank Balances', labelNep: 'नगद तथा बैंक शेष' },
   585→    govtSecurities: { weight: 0, label: 'Government Securities', labelNep: 'सरकारी प्रतिभूति' },
   586→    performingLoans: { weight: 100, label: 'Performing Loans', labelNep: 'सामान्य ऋण' },
   587→    watchListLoans: { weight: 100, label: 'Watch List Loans', labelNep: 'सतर्कता ऋण' },
   588→    subStandardLoans: { weight: 100, label: 'Sub-standard Loans', labelNep: 'उप-मानक ऋण' },
   589→    doubtfulLoans: { weight: 100, label: 'Doubtful Loans', labelNep: 'शंकास्पद ऋण' },
   590→    lossLoans: { weight: 100, label: 'Loss Loans', labelNep: 'घाटा ऋण' },
   591→    fixedAssets: { weight: 100, label: 'Fixed Assets', labelNep: 'स्थायी सम्पत्ति' },
   592→    otherAssets: { weight: 100, label: 'Other Assets', labelNep: 'अन्य सम्पत्ति' },
   593→  }
   594→
   595→  const totalFixedAssets = data.assets.reduce((s, a) => s + a.currentValue, 0)
   596→  const totalRiskWeightedAssets = totalOutstanding + totalFixedAssets // simplified: all at 100% risk weight
   597→
   598→  const capitalAdequacyRatio = totalRiskWeightedAssets > 0 ? (totalCapital / totalRiskWeightedAssets) * 100 : 0
   599→
   600→  return {
   601→    reportType: 'capital-adequacy',
   602→    reportTitle: 'Capital Adequacy Report',
   603→    reportTitleNep: 'पूँजी पर्याप्तता विवरण',
   604→    header,
   605→    capitalStructure: {
   606→      tier1Capital: {
   607→        paidUpCapital: { amount: totalShareCapital, label: 'Paid-up Share Capital', labelNep: 'चुक्ता शेयर पूँजी' },
   608→        reserveFund: { amount: netIncome > 0 ? Math.round(netIncome * 0.25) : 0, label: 'Statutory Reserve Fund', labelNep: 'सांविधिक आरक्ष कोष' },
   609→        retainedEarnings: { amount: netIncome > 0 ? Math.round(netIncome) : 0, label: 'Retained Earnings', labelNep: 'सञ्चित आय' },
   610→        totalTier1: { amount: Math.round(tier1Capital), label: 'Total Tier 1 Capital', labelNep: 'कुल तह १ पूँजी' },
   611→      },
   612→      tier2Capital: {
   613→        subordinatedDebt: { amount: 0, label: 'Subordinated Debt', labelNep: 'अधीनस्थ ऋण' },
   614→        generalProvision: { amount: 0, label: 'General Provision (excess)', labelNep: 'सामान्य बन्दोबस्त (अतिरिक्त)' },
   615→        totalTier2: { amount: tier2Capital, label: 'Total Tier 2 Capital', labelNep: 'कुल तह २ पूँजी' },
   616→      },
   617→      totalCapital: { amount: Math.round(totalCapital), label: 'Total Capital Fund', labelNep: 'कुल पूँजी कोष' },
   618→    },
   619→    riskWeightedAssets: {
   620→      items: Object.entries(riskWeights).map(([key, val]) => {
   621→        let amount = 0
   622→        if (key === 'performingLoans') amount = totalOutstanding
   623→        else if (key === 'fixedAssets') amount = totalFixedAssets
   624→        else if (key === 'otherAssets') amount = 0
   625→        return {
   626→          ...val,
   627→          grossAmount: amount,
   628→          riskWeight: val.weight + '%',
   629→          riskWeightedAmount: (amount * val.weight) / 100,
   630→        }
   631→      }),
   632→      totalRiskWeightedAssets: { amount: Math.round(totalRiskWeightedAssets), label: 'Total Risk-Weighted Assets', labelNep: 'कुल जोखिम-भारित सम्पत्ति' },
   633→    },
   634→    capitalAdequacyRatio: {
   635→      current: capitalAdequacyRatio.toFixed(2) + '%',
   636→      minimumRequired: NRB_COMPLIANCE.MIN_CAPITAL_ADEQUACY_RATIO + '%',
   637→      surplus: Math.max(0, capitalAdequacyRatio - NRB_COMPLIANCE.MIN_CAPITAL_ADEQUACY_RATIO).toFixed(2) + '%',
   638→      deficit: Math.max(0, NRB_COMPLIANCE.MIN_CAPITAL_ADEQUACY_RATIO - capitalAdequacyRatio).toFixed(2) + '%',
   639→      isCompliant: capitalAdequacyRatio >= NRB_COMPLIANCE.MIN_CAPITAL_ADEQUACY_RATIO,
   640→      requiredCapital: Math.round((totalRiskWeightedAssets * NRB_COMPLIANCE.MIN_CAPITAL_ADEQUACY_RATIO) / 100),
   641→      currentCapital: Math.round(totalCapital),
   642→    },
   643→    currency: 'NPR',
   644→    currencyNep: 'नेपाली रूपैयाँ',
   645→  }
   646→}

   649→// 5. Loan Portfolio Report (ऋण विवरण)
   650→// -----------------------------------------------------------
   651→
   652→function generateLoanPortfolio(data: Awaited<ReturnType<typeof fetchReportData>>) {
   653→  const header = getInstitutionHeader(data.organization)
   654→
   655→  const disbursedLoans = data.loanApplications.filter((l) => l.status === 'DISBURSED')
   656→  const totalDisbursed = disbursedLoans.reduce((s, l) => s + (l.disbursedAmount || 0), 0)
   657→  const totalOutstanding = disbursedLoans.reduce((s, l) => s + (l.outstandingAmount || 0), 0)
   658→  const totalCollected = totalDisbursed - totalOutstanding
   659→
   660→  // By product type
   661→  const byProduct = disbursedLoans.reduce((acc: Record<string, { count: number; disbursed: number; outstanding: number; interestRate: number }>, l) => {
   662→    const key = l.product?.name || 'Unknown'
   663→    if (!acc[key]) acc[key] = { count: 0, disbursed: 0, outstanding: 0, interestRate: 0 }
   664→    acc[key].count++
   665→    acc[key].disbursed += l.disbursedAmount || 0
   666→    acc[key].outstanding += l.outstandingAmount || 0
   667→    acc[key].interestRate = l.product?.interestRate || 0
   668→    return acc
   669→  }, {})
   670→
   671→  // By purpose
   672→  const byPurpose = disbursedLoans.reduce((acc: Record<string, { count: number; amount: number }>, l) => {
   673→    const key = l.purpose || 'General'
   674→    if (!acc[key]) acc[key] = { count: 0, amount: 0 }
   675→    acc[key].count++
   676→    acc[key].amount += l.disbursedAmount || 0
   677→    return acc
   678→  }, {})
   679→
   680→  // NPL classification
   681→  const nplBreakdown: Record<string, { count: number; amount: number; provision: number }> = {
   682→    PASS: { count: 0, amount: 0, provision: 0 },
   683→    WATCH_LIST: { count: 0, amount: 0, provision: 0 },
   684→    SUB_STANDARD: { count: 0, amount: 0, provision: 0 },
   685→    DOUBTFUL: { count: 0, amount: 0, provision: 0 },
   686→    LOSS: { count: 0, amount: 0, provision: 0 },
   687→  }
   688→
   689→  disbursedLoans.forEach((loan) => {
   690→    const overdueRepayments = loan.repayments.filter((r) => r.status === 'OVERDUE' || r.status === 'PENDING')
   691→    if (overdueRepayments.length > 0) {
   692→      const earliestOverdue = overdueRepayments.sort((a, b) => a.paymentDate.localeCompare(b.paymentDate))[0]
   693→      const daysPastDue = getDaysPastDue(earliestOverdue.paymentDate)
   694→      const classification = classifyNPL(daysPastDue)
   695→      const outstanding = loan.outstandingAmount || 0
   696→      const provisionRate = NRB_COMPLIANCE.NPL_CLASSIFICATION[classification].provision / 100
   697→
   698→      nplBreakdown[classification].count++
   699→      nplBreakdown[classification].amount += outstanding
   700→      nplBreakdown[classification].provision += outstanding * provisionRate
   701→    } else {
   702→      const outstanding = loan.outstandingAmount || 0
   703→      nplBreakdown.PASS.count++
   704→      nplBreakdown.PASS.amount += outstanding
   705→      nplBreakdown.PASS.provision += outstanding * 0.01
   706→    }
   707→  })
   708→
   709→  // By member (top borrowers)
   710→  const memberLoans = disbursedLoans.reduce((acc: Record<string, { name: string; outstanding: number; count: number }>, l) => {
   711→    const memberId = l.memberId
   712→    const name = `${l.member?.firstName || ''} ${l.member?.lastName || ''}`
   713→    if (!acc[memberId]) acc[memberId] = { name, outstanding: 0, count: 0 }
   714→    acc[memberId].outstanding += l.outstandingAmount || 0
   715→    acc[memberId].count++
   716→    return acc
   717→  }, {})
   718→
   719→  const topBorrowers = Object.values(memberLoans)
   720→    .sort((a, b) => b.outstanding - a.outstanding)
   721→    .slice(0, 20)
   722→
   723→  // Guarantor analysis
   724→  const guaranteedLoans = disbursedLoans.filter((l) => l.guarantorId)
   725→  const totalGuaranteedAmount = guaranteedLoans.reduce((s, l) => s + (l.outstandingAmount || 0), 0)
   726→
   727→  // Collection rate
   728→  const collectionRate = totalDisbursed > 0 ? ((totalCollected / totalDisbursed) * 100).toFixed(1) : '0.0'
   729→
   730→  // Overdue loans detail
   731→  const overdueLoans = disbursedLoans
   732→    .filter((l) => l.repayments.some((r) => r.status === 'OVERDUE' || r.status === 'PENDING'))
   733→    .map((l) => {
   734→      const overdueRepayments = l.repayments.filter((r) => r.status === 'OVERDUE' || r.status === 'PENDING')
   735→      const daysPastDue = overdueRepayments.length > 0
   736→        ? getDaysPastDue(overdueRepayments.sort((a, b) => a.paymentDate.localeCompare(b.paymentDate))[0].paymentDate)
   737→        : 0
   738→      return {
   739→        applicationNo: l.applicationNo,
   740→        memberName: `${l.member?.firstName || ''} ${l.member?.lastName || ''}`,
   741→        disbursedAmount: l.disbursedAmount || 0,
   742→        outstandingAmount: l.outstandingAmount || 0,
   743→        daysPastDue,
   744→        nplClassification: NRB_COMPLIANCE.NPL_CLASSIFICATION[classifyNPL(daysPastDue)].label,
   745→        nplClassificationNep: NRB_COMPLIANCE.NPL_CLASSIFICATION[classifyNPL(daysPastDue)].labelNep,
   746→        provisionRequired: (l.outstandingAmount || 0) * NRB_COMPLIANCE.NPL_CLASSIFICATION[classifyNPL(daysPastDue)].provision / 100,
   747→        nextDueDate: l.nextDueDate || 'N/A',
   748→      }
   749→    })
   750→
   751→  return {
   752→    reportType: 'loan-portfolio',
   753→    reportTitle: 'Loan Portfolio Report',
   754→    reportTitleNep: 'ऋण विवरण',
   755→    header,
   756→    summary: {
   757→      totalDisbursed: { amount: Math.round(totalDisbursed), label: 'Total Disbursed', labelNep: 'कुल वितरण' },
   758→      totalOutstanding: { amount: Math.round(totalOutstanding), label: 'Total Outstanding', labelNep: 'कुल बक्यौता' },
   759→      totalCollected: { amount: Math.round(totalCollected), label: 'Total Collected', labelNep: 'कुल संकलन' },
   760→      collectionRate: { value: collectionRate + '%', label: 'Collection Rate', labelNep: 'संकलन दर' },
   761→      totalLoanAccounts: { value: disbursedLoans.length, label: 'Total Loan Accounts', labelNep: 'कुल ऋण खाता' },
   762→      guaranteedLoans: { count: guaranteedLoans.length, amount: Math.round(totalGuaranteedAmount) },
   763→    },
   764→    byProduct: Object.entries(byProduct).map(([name, val]) => ({
   765→      productName: name,
   766→      count: val.count,
   767→      disbursed: Math.round(val.disbursed),
   768→      outstanding: Math.round(val.outstanding),
   769→      interestRate: val.interestRate + '%',
   770→    })),
   771→    byPurpose: Object.entries(byPurpose).map(([purpose, val]) => ({
   772→      purpose,
   773→      count: val.count,
   774→      amount: Math.round(val.amount),
   775→    })),
   776→    nplAnalysis: {
   777→      classification: Object.entries(nplBreakdown).map(([key, val]) => ({
   778→        classification: NRB_COMPLIANCE.NPL_CLASSIFICATION[key as keyof typeof NRB_COMPLIANCE.NPL_CLASSIFICATION].label,

   814→    acc[key].balance += sa.balance
   815→    acc[key].interest += sa.interestEarned
   816→    return acc

   823→    acc[key].count++
   824→    acc[key].balance += sa.balance
   825→    return acc

   841→  const totalWithdrawals = data.savingsTransactions.filter((t) => t.type === 'WITHDRAWAL').reduce((s, t) => s + t.amount, 0)

   849→  const closedAccounts = data.savingsAccounts.filter((sa) => sa.status === 'CLOSED').length
   850→  const frozenAccounts = data.savingsAccounts.filter((sa) => sa.status === 'FROZEN').length

   886→    currencyNep: 'नेपाली रूपैयाँ',

  1008→function generateCashFlow(data: Awaited<ReturnType<typeof fetchReportData>>) {
  1009→  const header = getInstitutionHeader(data.organization)
  1010→
  1011→  const totalSavingsDeposits = data.savingsAccounts.reduce((s, a) => s + a.balance, 0)
  1012→  const totalShareCapital = data.shareHoldings.reduce((s, sh) => s + sh.shareValue, 0)
  1013→  const totalLoansDisbursed = data.loanApplications
  1014→    .filter((l) => l.status === 'DISBURSED')
  1015→    .reduce((s, l) => s + (l.disbursedAmount || 0), 0)
  1016→  const totalOutstanding = data.loanApplications
  1017→    .filter((l) => l.status === 'DISBURSED')
  1018→    .reduce((s, l) => s + (l.outstandingAmount || 0), 0)
  1019→
  1020→  const interestIncome = data.loanApplications
  1021→    .filter((l) => l.status === 'DISBURSED')
  1022→    .reduce((s, l) => s + ((l.disbursedAmount || 0) * (l.interestRate || 0)) / 100 / 12, 0)
  1023→  const interestExpense = data.savingsAccounts.reduce((s, a) => s + (a.balance * (a.product?.interestRate || 0)) / 100 / 12, 0)
  1024→  const totalSalaryExpense = data.employees.reduce((s, e) => s + e.salary, 0)
  1025→
  1026→  // Inflows
  1027→  const depositInflows = data.savingsTransactions.filter((t) => t.type === 'DEPOSIT').reduce((s, t) => s + t.amount, 0)
  1028→  const loanRepaymentInflows = totalLoansDisbursed - totalOutstanding
  1029→  const shareCapitalInflows = totalShareCapital
  1030→  const feeIncome = data.loanApplications
  1031→    .filter((l) => l.status !== 'PENDING')
  1032→    .reduce((s, l) => s + (l.product?.processingFee || 0), 0)
  1033→  const penaltyIncome = data.loanRepayments.reduce((s, r) => s + r.penaltyAmount, 0)
  1034→  const totalInflows = depositInflows + loanRepaymentInflows + shareCapitalInflows + feeIncome + penaltyIncome
  1035→
  1036→  // Outflows
  1037→  const loanDisbursementOutflows = totalLoansDisbursed
  1038→  const withdrawalOutflows = data.savingsTransactions.filter((t) => t.type === 'WITHDRAWAL').reduce((s, t) => s + t.amount, 0)
  1039→  const salaryOutflows = totalSalaryExpense
  1040→  const interestPaymentOutflows = data.savingsTransactions.filter((t) => t.type === 'INTEREST').reduce((s, t) => s + t.amount, 0)
  1041→  const assetPurchaseOutflows = data.assets.reduce((s, a) => s + a.purchasePrice, 0)
  1042→  const totalOutflows = loanDisbursementOutflows + withdrawalOutflows + salaryOutflows + interestPaymentOutflows + assetPurchaseOutflows
  1043→
  1044→  // Net cash flow
  1045→  const netCashFlow = totalInflows - totalOutflows
  1046→
  1047→  // Operating activities
  1048→  const operatingInflows = {
  1049→    interestReceived: { amount: Math.round(loanRepaymentInflows), label: 'Interest & Principal Received', labelNep: 'ब्याज तथा मूल रकम प्राप्त' },
  1050→    feeIncome: { amount: Math.round(feeIncome), label: 'Fee Income', labelNep: 'शुल्क आय' },
  1051→    penaltyIncome: { amount: Math.round(penaltyIncome), label: 'Penalty Income', labelNep: 'जरिवाना आय' },
  1052→    totalOperatingInflows: { amount: Math.round(loanRepaymentInflows + feeIncome + penaltyIncome), label: 'Total Operating Inflows', labelNep: 'कुल सञ्चालन आगमन' },
  1053→  }
  1054→  const operatingOutflows = {
  1055→    interestPaid: { amount: Math.round(interestPaymentOutflows), label: 'Interest Paid on Deposits', labelNep: 'निक्षेपमा ब्याज भुक्तानी' },
  1056→    salaryPaid: { amount: Math.round(salaryOutflows), label: 'Salary & Benefits Paid', labelNep: 'तलब तथा भत्ता भुक्तानी' },
  1057→    officeExpenses: { amount: 0, label: 'Office & Administrative Expenses', labelNep: 'कार्यालय तथा प्रशासनिक खर्च' },
  1058→    totalOperatingOutflows: { amount: Math.round(interestPaymentOutflows + salaryOutflows), label: 'Total Operating Outflows', labelNep: 'कुल सञ्चालन जाने' },
  1059→  }
  1060→  const netOperatingCash = operatingInflows.totalOperatingInflows.amount - operatingOutflows.totalOperatingOutflows.amount
  1061→
  1062→  // Investing activities
  1063→  const investingInflows = {
  1064→    assetSale: { amount: 0, label: 'Proceeds from Asset Sale', labelNep: 'सम्पत्ति बिक्रीबाट प्राप्त' },
  1065→    totalInvestingInflows: { amount: 0, label: 'Total Investing Inflows', labelNep: 'कुल लगानी आगमन' },
  1066→  }
  1067→  const investingOutflows = {
  1068→    loanDisbursement: { amount: Math.round(loanDisbursementOutflows), label: 'Loan Disbursements', labelNep: 'ऋण वितरण' },
  1069→    assetPurchase: { amount: Math.round(assetPurchaseOutflows), label: 'Asset Purchases', labelNep: 'सम्पत्ति खरिद' },
  1070→    totalInvestingOutflows: { amount: Math.round(loanDisbursementOutflows + assetPurchaseOutflows), label: 'Total Investing Outflows', labelNep: 'कुल लगानी जाने' },
  1071→  }
  1072→  const netInvestingCash = investingInflows.totalInvestingInflows.amount - investingOutflows.totalInvestingOutflows.amount
  1073→
  1074→  // Financing activities
  1075→  const financingInflows = {
  1076→    memberDeposits: { amount: Math.round(depositInflows), label: 'Member Deposits Received', labelNep: 'सदस्य निक्षेप प्राप्त' },
  1077→    shareCapital: { amount: Math.round(shareCapitalInflows), label: 'Share Capital Received', labelNep: 'शेयर पूँजी प्राप्त' },
  1078→    totalFinancingInflows: { amount: Math.round(depositInflows + shareCapitalInflows), label: 'Total Financing Inflows', labelNep: 'कुल वित्तीय आगमन' },
  1079→  }
  1080→  const financingOutflows = {
  1081→    depositWithdrawals: { amount: Math.round(withdrawalOutflows), label: 'Deposit Withdrawals', labelNep: 'निक्षेप निकासा' },
  1082→    shareRedemptions: { amount: 0, label: 'Share Redemptions', labelNep: 'शेयर फिर्ता' },
  1083→    totalFinancingOutflows: { amount: Math.round(withdrawalOutflows), label: 'Total Financing Outflows', labelNep: 'कुल वित्तीय जाने' },
  1084→  }
  1085→  const netFinancingCash = financingInflows.totalFinancingInflows.amount - financingOutflows.totalFinancingOutflows.amount
  1086→
  1087→  return {
  1088→    reportType: 'cash-flow',
  1089→    reportTitle: 'Cash Flow Statement',
  1090→    reportTitleNep: 'नगद प्रवाह विवरण',
  1091→    header,
  1092→    operatingActivities: {
  1093→      inflows: operatingInflows,
  1094→      outflows: operatingOutflows,
  1095→      netCashFromOperations: { amount: Math.round(netOperatingCash), label: 'Net Cash from Operations', labelNep: 'सञ्चालनबाट खुद नगद' },
  1096→    },
  1097→    investingActivities: {
  1098→      inflows: investingInflows,
  1099→      outflows: investingOutflows,
  1100→      netCashFromInvesting: { amount: Math.round(netInvestingCash), label: 'Net Cash from Investing', labelNep: 'लगानीबाट खुद नगद' },
  1101→    },
  1102→    financingActivities: {
  1103→      inflows: financingInflows,
  1104→      outflows: financingOutflows,
  1105→      netCashFromFinancing: { amount: Math.round(netFinancingCash), label: 'Net Cash from Financing', labelNep: 'वित्तीयबाट खुद नगद' },
  1106→    },
  1107→    summary: {
  1108→      totalInflows: { amount: Math.round(totalInflows), label: 'Total Cash Inflows', labelNep: 'कुल नगद आगमन' },
  1109→      totalOutflows: { amount: Math.round(totalOutflows), label: 'Total Cash Outflows', labelNep: 'कुल नगद जाने' },
  1110→      netCashFlow: { amount: Math.round(netCashFlow), label: 'Net Cash Flow', labelNep: 'खुद नगद प्रवाह' },
  1111→      cashAtBeginning: { amount: 0, label: 'Cash at Beginning of Period', labelNep: 'अवधि सुरुमा नगद' },
  1112→      cashAtEnd: { amount: Math.round(netCashFlow), label: 'Cash at End of Period', labelNep: 'अवधि अन्त्यमा नगद' },
  1113→    },
  1114→    currency: 'NPR',
  1115→    currencyNep: 'नेपाली रूपैयाँ',
  1116→  }

  1147→export async function GET(request: Request) {
  1148→  try {
  1149→    const { searchParams } = new URL(request.url)
  1150→    const reportType = searchParams.get('type') as ReportType | null
  1151→    const format = (searchParams.get('format') || 'json').toLowerCase()
  1152→
  1153→    // Validate report type
  1154→    if (!reportType || !VALID_REPORT_TYPES.includes(reportType)) {
  1155→      return NextResponse.json(
  1156→        {
  1157→          error: 'Invalid or missing report type',
  1158→          validTypes: VALID_REPORT_TYPES,
  1159→          usage: '/api/reports/nrb?type=<report-type>&format=<json|pdf>',
  1160→        },
  1161→        { status: 400 }
  1162→      )
  1163→    }
  1164→
  1165→    // Validate format
  1166→    if (format !== 'json' && format !== 'pdf') {
  1167→      return NextResponse.json(
  1168→        {
  1169→          error: 'Invalid format. Use "json" or "pdf"',
  1170→        },
  1171→        { status: 400 }
  1172→      )
