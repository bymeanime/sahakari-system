   105→function BSDatePicker({ value, onChange, label }: { value: string; onChange: (v: string) => void; label?: string }) {
   106→  const [open, setOpen] = useState(false)
   107→  const parts = value ? value.split('-').map(Number) : [2082, 1, 1]
   108→  const [year, setYear] = useState(parts[0])
   109→  const [month, setMonth] = useState(parts[1])
   110→  const [day, setDay] = useState(parts[2])
   111→  const years = getBSYearRange()
   112→
   113→  const formatted = value ? formatBSDate(value) : { en: 'Select Date', nep: 'मिति छान्नुहोस्' }
   114→  const grid = getBSMonthGrid(year, month)
   115→  const maxDays = getBSMonthDays(year, month)
   116→
   117→  const selectDay = (d: number) => {
   118→    const dateStr = `${year}-${String(month).padStart(2, '0')}-${String(d).padStart(2, '0')}`
   119→    onChange(dateStr)
   120→    setOpen(false)
   121→  }
   122→
   123→  return (
   124→    <div className="space-y-1">
   125→      {label && <Label className="text-xs">{label}</Label>}
   126→      <div className="relative">
   127→        <button
   128→          onClick={() => setOpen(!open)}
   129→          className="w-full flex items-center justify-between px-3 py-2 border rounded-lg bg-white text-sm text-left hover:border-emerald-400 transition-colors"
   130→        >
   131→          <div>
   132→            <span className="text-gray-900">{formatted.en}</span>
   133→            <span className="text-gray-400 ml-2 text-xs">({formatted.nep})</span>
   134→          </div>
   135→          <CalendarDays className="w-4 h-4 text-gray-400" />
   136→        </button>
   137→        {open && (
   138→          <div className="absolute top-full mt-1 left-0 z-50 bg-white border rounded-xl shadow-xl p-4 w-80">
   139→            <div className="flex items-center gap-2 mb-3">
   140→              <Select value={String(year)} onValueChange={v => { setYear(Number(v)); setDay(1) }}>
   141→                <SelectTrigger className="w-24 h-8 text-xs"><SelectValue /></SelectTrigger>
   142→                <SelectContent>
   143→                  {years.map(y => <SelectItem key={y} value={String(y)}>{toNepaliDigits(y)}</SelectItem>)}
   144→                </SelectContent>
   145→              </Select>
   146→              <Select value={String(month)} onValueChange={v => { setMonth(Number(v)); setDay(1) }}>
   147→                <SelectTrigger className="w-28 h-8 text-xs"><SelectValue /></SelectTrigger>
   148→                <SelectContent>
   149→                  {bsMonths.map((m, i) => <SelectItem key={i} value={String(i + 1)}>{m} ({bsMonthsNep[i]})</SelectItem>)}
   150→                </SelectContent>
   151→              </Select>
   152→            </div>
   153→            <div className="grid grid-cols-7 gap-0.5 text-center mb-1">
   154→              {['आ', 'सो', 'मं', 'बु', 'बि', 'शु', 'श'].map((d, i) => (
   155→                <div key={i} className="text-xs text-gray-400 py-1 font-medium">{d}</div>
   156→              ))}
   157→            </div>
   158→            <div className="grid grid-cols-7 gap-0.5">
   159→              {grid.flat().map((d, i) => (
   160→                <button
   161→                  key={i}
   162→                  disabled={!d}
   163→                  onClick={() => d && selectDay(d)}
   164→                  className={`h-8 w-8 rounded-lg text-xs flex items-center justify-center transition-colors ${
   165→                    d === day && month === parts[1] && year === parts[0]
   166→                      ? 'bg-emerald-600 text-white font-bold'
   167→                      : d ? 'hover:bg-emerald-50 text-gray-700' : 'text-transparent'
   168→                  }`}
   169→                >
   170→                  {d ? toNepaliDigits(d) : ''}
   171→                </button>
   172→              ))}
   173→            </div>
   174→            <div className="mt-3 pt-2 border-t flex justify-between items-center">
   175→              <span className="text-xs text-gray-400">{bsMonths[month - 1]} {year} - {toNepaliDigits(maxDays)} days</span>
   176→              <Button size="sm" variant="ghost" className="text-xs h-7" onClick={() => { onChange(getTodayBS()); setOpen(false) }}>Today</Button>
   177→            </div>
   178→          </div>
   179→        )}
   180→      </div>
   181→    </div>
   182→  )
   183→}

   188→function NotificationPanel({ open, onClose }: { open: boolean; onClose: () => void }) {
   189→  const [notifications, setNotifications] = useState<any[]>([])
   190→  const [loading, setLoading] = useState(false)
   191→
   192→  useEffect(() => {
   193→    if (open) {
   194→      fetch('/api/notifications')
   195→        .then(r => r.json())
   196→        .then(d => { setNotifications(Array.isArray(d) ? d : []) })
   197→        .catch(() => {})
   198→    }
   199→  }, [open])
   200→
   201→  if (!open) return null
   202→
   203→  const severityIcon = (s: string) => {
   204→    if (s === 'ERROR') return <AlertCircle className="w-4 h-4 text-red-500" />
   205→    if (s === 'WARNING') return <AlertTriangle className="w-4 h-4 text-amber-500" />
   206→    if (s === 'SUCCESS') return <CheckCircle className="w-4 h-4 text-emerald-500" />
   207→    return <Info className="w-4 h-4 text-blue-500" />
   208→  }
   209→
   210→  return (
   211→    <div className="fixed inset-0 z-50 flex justify-end">
   212→      <div className="absolute inset-0 bg-black/30" onClick={onClose} />
   213→      <div className="relative w-full max-w-md bg-white shadow-2xl h-full overflow-y-auto">
   214→        <div className="sticky top-0 bg-white border-b px-4 py-3 flex items-center justify-between z-10">
   215→          <div>
   216→            <h3 className="font-semibold text-gray-900">Notifications</h3>
   217→            <p className="text-xs text-gray-500">सूचना तथा चेतावनी</p>
   218→          </div>
   219→          <button onClick={onClose} className="p-1 rounded hover:bg-gray-100"><X className="w-5 h-5" /></button>
   220→        </div>
   221→        <div className="p-4 space-y-3">
   222→          {loading ? (
   223→            <div className="flex items-center justify-center py-12"><Loader2 className="w-6 h-6 animate-spin text-emerald-600" /></div>
   224→          ) : notifications.length === 0 ? (
   225→            <div className="text-center py-12 text-gray-400"><Bell className="w-12 h-12 mx-auto mb-3 opacity-50" /><p>No notifications</p></div>
   226→          ) : (
   227→            notifications.map((n, i) => (
   228→              <div key={n.id || i} className={`p-3 rounded-lg border ${
   229→                n.severity === 'ERROR' ? 'border-red-200 bg-red-50' :
   230→                n.severity === 'WARNING' ? 'border-amber-200 bg-amber-50' :
   231→                n.severity === 'SUCCESS' ? 'border-emerald-200 bg-emerald-50' :
   232→                'border-blue-200 bg-blue-50'
   233→              }`}>
   234→                <div className="flex items-start gap-2">
   235→                  {severityIcon(n.severity)}
   236→                  <div className="flex-1 min-w-0">
   237→                    <p className="text-sm font-medium text-gray-900">{n.title}</p>
   238→                    <p className="text-xs text-gray-600 mt-0.5">{n.message}</p>
   239→                    <p className="text-xs text-gray-400 mt-0.5">{n.messageNep}</p>
   240→                    {n.action === 'SEND_SMS' && (
   241→                      <Button size="sm" variant="outline" className="mt-2 h-7 text-xs" onClick={async () => {
   242→                        try {
   243→                          const res = await fetch('/api/sms', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ type: 'emi-reminder', loanId: n.id?.replace('emi-', '') }) })
   244→                          if (res.ok) toast.success('SMS sent to member')
   245→                          else toast.error('Failed to send SMS')
   246→                        } catch { toast.error('SMS service unavailable') }
   247→                      }}>
   248→                        <Send className="w-3 h-3 mr-1" /> Send SMS
   249→                      </Button>
   250→                    )}
   251→                  </div>
   252→                </div>
   253→              </div>
   254→            ))
   255→          )}
   256→        </div>
   257→      </div>
   258→    </div>
   259→  )
