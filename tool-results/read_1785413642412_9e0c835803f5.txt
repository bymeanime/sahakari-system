    63→    63→function formatNPR(amount: number): string {
    64→    64→  return `Rs. ${amount.toLocaleString('en-NP', { minimumFractionDigits: 2, maximumFractionDigits: 2 })}`
    65→    65→}
    66→    66→
    67→    67→function toNepaliDigits(num: number | string): string {
    68→    68→  const nepaliDigits = ['०', '१', '२', '३', '४', '५', '६', '७', '८', '९']
    69→    69→  return String(num).replace(/[0-9]/g, (d) => nepaliDigits[parseInt(d)])
    70→    70→}
    71→    71→
    72→    72→function formatNPRNepali(amount: number): string {
    73→    73→  const formatted = amount.toLocaleString('en-NP', { minimumFractionDigits: 2, maximumFractionDigits: 2 })
    74→    74→  return `रु. ${toNepaliDigits(formatted)}`
    75→    75→}

    77→    77→/** Get current fiscal year in BS format */
    78→    78→function getCurrentFiscalYear(): string {
    79→    79→  const today = new Date()
    80→    80→  const bsYear = today.getFullYear() + 57
    81→    81→  // BS fiscal year: if current month is after Chaitra (mid-April), next FY
    82→    82→  const fiscalStart = today.getMonth() + 1 > 3 ? bsYear : bsYear - 1
    83→    83→  return `${fiscalStart}/${(fiscalStart + 1) % 100}`
    84→    84→}
    85→    85→
    86→    86→/** Get previous fiscal year */
    87→    87→function getPreviousFiscalYear(): string {
    88→    88→  const current = getCurrentFiscalYear()
    89→    89→  const [startYear] = current.split('/').map(Number)
    90→    90→  return `${startYear - 1}/${startYear % 100}`
    91→    91→}

    93→    93→/** Classify NPL based on days past due */
    94→    94→function classifyNPL(daysPastDue: number): keyof typeof NRB_COMPLIANCE.NPL_CLASSIFICATION {
    95→    95→  if (daysPastDue <= NRB_COMPLIANCE.NPL_THRESHOLDS.PASS) return 'PASS'
    96→    96→  if (daysPastDue <= NRB_COMPLIANCE.NPL_THRESHOLDS.WATCH_LIST) return 'WATCH_LIST'
    97→    97→  if (daysPastDue <= NRB_COMPLIANCE.NPL_THRESHOLDS.SUB_STANDARD) return 'SUB_STANDARD'
    98→    98→  if (daysPastDue <= NRB_COMPLIANCE.NPL_THRESHOLDS.DOUBTFUL) return 'DOUBTFUL'
    99→    99→  return 'LOSS'
   100→   100→}

   102→   102→/** Calculate days past due from a due date string */
   103→   103→function getDaysPastDue(dueDate: string): number {
   104→   104→  const due = new Date(dueDate)
   105→   105→  const now = new Date()
   106→   106→  const diffMs = now.getTime() - due.getTime()
   107→   107→  return Math.max(0, Math.floor(diffMs / (1000 * 60 * 60 * 24)))
   108→   108→}

   227→   227→function generateBalanceSheet(data: Awaited<ReturnType<typeof fetchReportData>>) {
   228→   228→  const header = getInstitutionHeader(data.organization)
   229→   229→
   230→   230→  // Calculate totals
   231→   231→  const totalSavingsDeposits = data.savingsAccounts.reduce((s, a) => s + a.balance, 0)
   232→   232→  const totalShareCapital = data.shareHoldings.reduce((s, sh) => s + sh.shareValue, 0)
   233→   233→  const totalLoanReceivable = data.loanApplications
   234→   234→    .filter((l) => l.status === 'DISBURSED')
   235→   235→    .reduce((s, l) => s + (l.outstandingAmount || 0), 0)
   236→   236→  const totalFixedAssets = data.assets.reduce((s, a) => s + a.currentValue, 0)
   237→   237→  const totalDepreciation = data.assets.reduce((s, a) => s + a.accumulatedDep, 0)
   238→   238→  const netFixedAssets = totalFixedAssets - totalDepreciation
   239→   239→
   240→   240→  // Accounting totals from journal entries
   241→   241→  const totalDebits = data.journalEntries.reduce((sum, je) => sum + je.items.reduce((s, i) => s + i.debit, 0), 0)
   242→   242→  const totalCredits = data.journalEntries.reduce((sum, je) => sum + je.items.reduce((s, i) => s + i.credit, 0), 0)
   243→   243→
   244→   244→  // Income & Expense for retained earnings
   245→   245→  const interestIncome = data.loanApplications
   246→   246→    .filter((l) => l.status === 'DISBURSED')
   247→   247→    .reduce((s, l) => s + ((l.disbursedAmount || 0) * (l.interestRate || 0)) / 100 / 12, 0)
   248→   248→  const interestExpense = data.savingsAccounts.reduce((s, a) => s + (a.balance * (a.product?.interestRate || 0)) / 100 / 12, 0)
   249→   249→  const totalSalaryExpense = data.employees.reduce((s, e) => s + e.salary, 0)
   250→   250→  const netIncome = interestIncome - interestExpense - totalSalaryExpense
   251→   251→
   252→   252→  // Cash & bank balance from accounts
   253→   253→  const cashAccounts = data.accounts.filter((a) => a.type === 'ASSET' && (a.subType === 'CASH' || a.subType === 'BANK'))
   254→   254→  const cashInHand = cashAccounts.filter((a) => a.subType === 'CASH').reduce((s, a) => {
   255→   255→    const debitTotal = data.journalEntryItems.filter((i) => i.accountId === a.id).reduce((s, i) => s + i.debit - i.credit, 0)
   256→   256→    return s + Math.max(0, debitTotal)
   257→   257→  }, 0)
   258→   258→  const bankBalance = cashAccounts.filter((a) => a.subType === 'BANK').reduce((s, a) => {
   259→   259→    const debitTotal = data.journalEntryItems.filter((i) => i.accountId === a.id).reduce((s, i) => s + i.debit - i.credit, 0)
   260→   260→    return s + Math.max(0, debitTotal)
   261→   261→  }, 0)
   262→   262→
   263→   263→  // Reserve fund (from equity accounts)
   264→   264→  const reserveFund = data.accounts
   265→   265→    .filter((a) => a.type === 'EQUITY' && a.name.toLowerCase().includes('reserve'))
   266→   266→    .reduce((s, a) => {
   267→   267→      const creditTotal = data.journalEntryItems.filter((i) => i.accountId === a.id).reduce((s, i) => s + i.credit - i.debit, 0)
   268→   268→      return s + Math.max(0, creditTotal)
   269→   269→    }, 0)
   270→   270→
   271→   271→  const totalAssets = cashInHand + bankBalance + totalLoanReceivable + netFixedAssets
   272→   272→  const totalLiabilities = totalSavingsDeposits + totalShareCapital + reserveFund + (netIncome > 0 ? netIncome : 0)
   273→   273→
   274→   274→  return {
   275→   275→    reportType: 'balance-sheet',
   276→   276→    reportTitle: 'Balance Sheet',
   277→   277→    reportTitleNep: 'निश्शेष जानकारी',
   278→   278→    header,
   279→   279→    nrbCompliance: {
   280→   280→      capitalAdequacyRatio: totalLoanReceivable > 0 ? ((totalShareCapital + reserveFund + (netIncome > 0 ? netIncome : 0)) / totalLoanReceivable) * 100 : 0,
   281→   281→      minRequiredRatio: NRB_COMPLIANCE.MIN_CAPITAL_ADEQUACY_RATIO,
   282→   282→      isCompliant: totalLoanReceivable > 0 ? ((totalShareCapital + reserveFund + (netIncome > 0 ? netIncome : 0)) / totalLoanReceivable) * 100 >= NRB_COMPLIANCE.MIN_CAPITAL_ADEQUACY_RATIO : true,
   283→   283→    },
   284→   284→    currentPeriod: {
   285→   285→      assets: {
   286→   286→        cashInHand: { amount: cashInHand, label: 'Cash in Hand', labelNep: 'हातमा नगद' },
   287→   287→        bankBalance: { amount: bankBalance, label: 'Bank Balance', labelNep: 'बैंक शेष' },
   288→   288→        loanReceivable: { amount: totalLoanReceivable, label: 'Loan Receivable', labelNep: 'ऋण उपलब्ध' },
   289→   289→        fixedAssets: { amount: netFixedAssets, label: 'Fixed Assets (Net)', labelNep: 'स्थायी सम्पत्ति (शुद्ध)' },
   290→   290→        totalAssets: { amount: totalAssets, label: 'Total Assets', labelNep: 'कुल सम्पत्ति' },
   291→   291→      },
   292→   292→      liabilities: {
   293→   293→        memberDeposits: { amount: totalSavingsDeposits, label: 'Member Deposits', labelNep: 'सदस्य निक्षेप' },
   294→   294→        shareCapital: { amount: totalShareCapital, label: 'Share Capital', labelNep: 'शेयर पूँजी' },
   295→   295→        reserveFund: { amount: reserveFund, label: 'Reserve Fund', labelNep: 'आरक्ष कोष' },
   296→   296→        retainedEarnings: { amount: netIncome > 0 ? netIncome : 0, label: 'Retained Earnings', labelNep: 'सञ्चित आय' },
   297→   297→        totalLiabilities: { amount: totalLiabilities, label: 'Total Liabilities & Equity', labelNep: 'कुल दायित्व र पूँजी' },
   298→   298→      },
   299→   299→    },
   300→   300→    previousPeriod: {
   301→   301→      // Previous year comparison (placeholder - would need historical data)
   302→   302→      note: 'Previous year figures require historical data from prior fiscal year records.',
   303→   303→    },
   304→   304→    summary: {
   305→   305→      totalAssets,
   306→   306→      totalLiabilities,
   307→   307→      surplus: totalAssets - totalLiabilities,
   308→   308→      isBalanced: Math.abs(totalAssets - totalLiabilities) < 1,
   309→   309→    },
   310→   310→    currency: 'NPR',
   311→   311→    currencyNep: 'नेपाली रूपैयाँ',
   312→   312→  }
   313→   313→}

   318→   318→
   319→   319→function generateIncomeStatement(data: Awaited<ReturnType<typeof fetchReportData>>) {
   320→   320→  const header = getInstitutionHeader(data.organization)
   321→   321→
   322→   322→  // Income calculations
   323→   323→  const interestIncome = data.loanApplications
   324→   324→    .filter((l) => l.status === 'DISBURSED')
   325→   325→    .reduce((s, l) => s + ((l.disbursedAmount || 0) * (l.interestRate || 0)) / 100 / 12, 0)
   326→   326→  const feeIncome = data.loanApplications
   327→   327→    .filter((l) => l.status !== 'PENDING')
   328→   328→    .reduce((s, l) => s + (l.product?.processingFee || 0), 0)
   329→   329→  const penaltyIncome = data.loanRepayments.reduce((s, r) => s + r.penaltyAmount, 0)
   330→   330→
   331→   331→  // Expense calculations
   332→   332→  const interestExpense = data.savingsAccounts.reduce((s, a) => s + (a.balance * (a.product?.interestRate || 0)) / 100 / 12, 0)
   333→   333→  const totalSalaryExpense = data.employees.reduce((s, e) => s + e.salary, 0)
   334→   334→  const totalDepreciation = data.assets.reduce((s, a) => s + a.accumulatedDep, 0) / 5
   335→   335→  const officeExpense = data.accounts
   336→   336→    .filter((a) => a.type === 'EXPENSE' && a.name.toLowerCase().includes('office'))
   337→   337→    .reduce((s, a) => {
   338→   338→      const debitTotal = data.journalEntryItems.filter((i) => i.accountId === a.id).reduce((s, i) => s + i.debit, 0)
   339→   339→      return s + debitTotal
   340→   340→    }, 0)
   341→   341→
   342→   342→  const totalIncome = interestIncome + feeIncome + penaltyIncome
   343→   343→  const totalExpenses = interestExpense + totalSalaryExpense + officeExpense + totalDepreciation
   344→   344→  const netIncome = totalIncome - totalExpenses
   345→   345→
   346→   346→  return {
   347→   347→    reportType: 'income-statement',
   348→   348→    reportTitle: 'Income Statement',
   349→   349→    reportTitleNep: 'आय-व्यय जानकारी',
   350→   350→    header,
   351→   351→    currentPeriod: {
   352→   352→      income: {
   353→   353→        interestIncome: { amount: Math.round(interestIncome), label: 'Interest Income', labelNep: 'ब्याज आय' },
   354→   354→        feeIncome: { amount: Math.round(feeIncome), label: 'Fee & Commission Income', labelNep: 'शुल्क तथा कमिसन आय' },
   355→   355→        penaltyIncome: { amount: Math.round(penaltyIncome), label: 'Penalty Income', labelNep: 'जरिवाना आय' },
   356→   356→        otherIncome: { amount: 0, label: 'Other Income', labelNep: 'अन्य आय' },
   357→   357→        totalIncome: { amount: Math.round(totalIncome), label: 'Total Income', labelNep: 'कुल आय' },
   358→   358→      },
   359→   359→      expenses: {
   360→   360→        interestExpense: { amount: Math.round(interestExpense), label: 'Interest Expense', labelNep: 'ब्याज खर्च' },
   361→   361→        salaryExpense: { amount: Math.round(totalSalaryExpense), label: 'Salary & Benefits', labelNep: 'तलब तथा भत्ता' },
   362→   362→        officeExpense: { amount: Math.round(officeExpense), label: 'Office & Administrative', labelNep: 'कार्यालय तथा प्रशासनिक' },
   363→   363→        depreciation: { amount: Math.round(totalDepreciation), label: 'Depreciation', labelNep: 'ह्रासकट्टी' },
   364→   364→        otherExpenses: { amount: 0, label: 'Other Expenses', labelNep: 'अन्य खर्च' },
   365→   365→        totalExpenses: { amount: Math.round(totalExpenses), label: 'Total Expenses', labelNep: 'कुल खर्च' },
   366→   366→      },
   367→   367→      netIncome: {
   368→   368→        amount: Math.round(netIncome),
   369→   369→        label: 'Net Income / (Loss)',
   370→   370→        labelNep: 'खुद आय / (घाटा)',
   371→   371→      },
   372→   372→    },
   373→   373→    previousPeriod: {
   374→   374→      note: 'Previous year figures require historical data from prior fiscal year records.',
   375→   375→    },
   376→   376→    keyRatios: {

   671→   671→    const key = l.purpose || 'General'
   672→   672→    if (!acc[key]) acc[key] = { count: 0, amount: 0 }
   673→   673→    acc[key].count++
   674→   674→    acc[key].amount += l.disbursedAmount || 0

   682→   682→    SUB_STANDARD: { count: 0, amount: 0, provision: 0 },
   683→   683→    DOUBTFUL: { count: 0, amount: 0, provision: 0 },
   684→   684→    LOSS: { count: 0, amount: 0, provision: 0 },
   685→   685→  }
   686→   686→
   687→   687→  disbursedLoans.forEach((loan) => {
   688→   688→    const overdueRepayments = loan.repayments.filter((r) => r.status === 'OVERDUE' || r.status === 'PENDING')
   689→   689→    if (overdueRepayments.length > 0) {
   690→   690→      const earliestOverdue = overdueRepayments.sort((a, b) => a.paymentDate.localeCompare(b.paymentDate))[0]
   691→   691→      const daysPastDue = getDaysPastDue(earliestOverdue.paymentDate)
   692→   692→      const classification = classifyNPL(daysPastDue)
   693→   693→      const outstanding = loan.outstandingAmount || 0
   694→   694→      const provisionRate = NRB_COMPLIANCE.NPL_CLASSIFICATION[classification].provision / 100
   695→   695→
   696→   696→      nplBreakdown[classification].count++
   697→   697→      nplBreakdown[classification].amount += outstanding
   698→   698→      nplBreakdown[classification].provision += outstanding * provisionRate
   699→   699→    } else {
   700→   700→      const outstanding = loan.outstandingAmount || 0
   701→   701→      nplBreakdown.PASS.count++
   702→   702→      nplBreakdown.PASS.amount += outstanding
   703→   703→      nplBreakdown.PASS.provision += outstanding * 0.01
   704→   704→    }
   705→   705→  })
   706→   706→
   707→   707→  // By member (top borrowers)
   708→   708→  const memberLoans = disbursedLoans.reduce((acc: Record<string, { name: string; outstanding: number; count: number }>, l) => {

   711→   711→    if (!acc[memberId]) acc[memberId] = { name, outstanding: 0, count: 0 }
   712→   712→    acc[memberId].outstanding += l.outstandingAmount || 0
   713→   713→    acc[memberId].count++
   714→   714→    return acc
   715→   715→  }, {})
   716→   716→
   717→   717→  const topBorrowers = Object.values(memberLoans)
   718→   718→    .sort((a, b) => b.outstanding - a.outstanding)
   719→   719→    .slice(0, 20)
   720→   720→
   721→   721→  // Guarantor analysis
   722→   722→  const guaranteedLoans = disbursedLoans.filter((l) => l.guarantorId)
   723→   723→  const totalGuaranteedAmount = guaranteedLoans.reduce((s, l) => s + (l.outstandingAmount || 0), 0)
   724→   724→
   725→   725→  // Collection rate
   726→   726→  const collectionRate = totalDisbursed > 0 ? ((totalCollected / totalDisbursed) * 100).toFixed(1) : '0.0'
   727→   727→
   728→   728→  // Overdue loans detail
   729→   729→  const overdueLoans = disbursedLoans
   730→   730→    .filter((l) => l.repayments.some((r) => r.status === 'OVERDUE' || r.status === 'PENDING'))
   731→   731→    .map((l) => {
   732→   732→      const overdueRepayments = l.repayments.filter((r) => r.status === 'OVERDUE' || r.status === 'PENDING')
   733→   733→      const daysPastDue = overdueRepayments.length > 0
   734→   734→        ? getDaysPastDue(overdueRepayments.sort((a, b) => a.paymentDate.localeCompare(b.paymentDate))[0].paymentDate)
   735→   735→        : 0
   736→   736→      return {
   737→   737→        applicationNo: l.applicationNo,
   738→   738→        memberName: `${l.member?.firstName || ''} ${l.member?.lastName || ''}`,
   739→   739→        disbursedAmount: l.disbursedAmount || 0,
   740→   740→        outstandingAmount: l.outstandingAmount || 0,
   741→   741→        daysPastDue,
   742→   742→        nplClassification: NRB_COMPLIANCE.NPL_CLASSIFICATION[classifyNPL(daysPastDue)].label,
   743→   743→        nplClassificationNep: NRB_COMPLIANCE.NPL_CLASSIFICATION[classifyNPL(daysPastDue)].labelNep,
   744→   744→        provisionRequired: (l.outstandingAmount || 0) * NRB_COMPLIANCE.NPL_CLASSIFICATION[classifyNPL(daysPastDue)].provision / 100,
   745→   745→        nextDueDate: l.nextDueDate || 'N/A',
   746→   746→      }
   747→   747→    })
   748→   748→
   749→   749→  return {
   750→   750→    reportType: 'loan-portfolio',
   751→   751→    reportTitle: 'Loan Portfolio Report',
   752→   752→    reportTitleNep: 'ऋण विवरण',
   753→   753→    header,
   754→   754→    summary: {
   755→   755→      totalDisbursed: { amount: Math.round(totalDisbursed), label: 'Total Disbursed', labelNep: 'कुल वितरण' },
   756→   756→      totalOutstanding: { amount: Math.round(totalOutstanding), label: 'Total Outstanding', labelNep: 'कुल बक्यौता' },
   757→   757→      totalCollected: { amount: Math.round(totalCollected), label: 'Total Collected', labelNep: 'कुल संकलन' },
   758→   758→      collectionRate: { value: collectionRate + '%', label: 'Collection Rate', labelNep: 'संकलन दर' },

   765→   765→      disbursed: Math.round(val.disbursed),
   766→   766→      outstanding: Math.round(val.outstanding),
   767→   767→      interestRate: val.interestRate + '%',
   768→   768→    })),

   772→   772→      amount: Math.round(val.amount),
   773→   773→    })),
   774→   774→    nplAnalysis: {
   775→   775→      classification: Object.entries(nplBreakdown).map(([key, val]) => ({

  1005→  1005→
  1006→  1006→function generateCashFlow(data: Awaited<ReturnType<typeof fetchReportData>>) {
  1007→  1007→  const header = getInstitutionHeader(data.organization)
  1008→  1008→
  1009→  1009→  const totalSavingsDeposits = data.savingsAccounts.reduce((s, a) => s + a.balance, 0)
  1010→  1010→  const totalShareCapital = data.shareHoldings.reduce((s, sh) => s + sh.shareValue, 0)
  1011→  1011→  const totalLoansDisbursed = data.loanApplications
  1012→  1012→    .filter((l) => l.status === 'DISBURSED')
  1013→  1013→    .reduce((s, l) => s + (l.disbursedAmount || 0), 0)
  1014→  1014→  const totalOutstanding = data.loanApplications
  1015→  1015→    .filter((l) => l.status === 'DISBURSED')
  1016→  1016→    .reduce((s, l) => s + (l.outstandingAmount || 0), 0)
  1017→  1017→
  1018→  1018→  const interestIncome = data.loanApplications
  1019→  1019→    .filter((l) => l.status === 'DISBURSED')
  1020→  1020→    .reduce((s, l) => s + ((l.disbursedAmount || 0) * (l.interestRate || 0)) / 100 / 12, 0)
  1021→  1021→  const interestExpense = data.savingsAccounts.reduce((s, a) => s + (a.balance * (a.product?.interestRate || 0)) / 100 / 12, 0)
  1022→  1022→  const totalSalaryExpense = data.employees.reduce((s, e) => s + e.salary, 0)
  1023→  1023→
  1024→  1024→  // Inflows
  1025→  1025→  const depositInflows = data.savingsTransactions.filter((t) => t.type === 'DEPOSIT').reduce((s, t) => s + t.amount, 0)
  1026→  1026→  const loanRepaymentInflows = totalLoansDisbursed - totalOutstanding
  1027→  1027→  const shareCapitalInflows = totalShareCapital
  1028→  1028→  const feeIncome = data.loanApplications
  1029→  1029→    .filter((l) => l.status !== 'PENDING')
  1030→  1030→    .reduce((s, l) => s + (l.product?.processingFee || 0), 0)
  1031→  1031→  const penaltyIncome = data.loanRepayments.reduce((s, r) => s + r.penaltyAmount, 0)
  1032→  1032→  const totalInflows = depositInflows + loanRepaymentInflows + shareCapitalInflows + feeIncome + penaltyIncome
  1033→  1033→
  1034→  1034→  // Outflows
  1035→  1035→  const loanDisbursementOutflows = totalLoansDisbursed
  1036→  1036→  const withdrawalOutflows = data.savingsTransactions.filter((t) => t.type === 'WITHDRAWAL').reduce((s, t) => s + t.amount, 0)
  1037→  1037→  const salaryOutflows = totalSalaryExpense
